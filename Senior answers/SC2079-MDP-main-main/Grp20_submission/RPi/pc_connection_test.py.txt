import logging
logging.basicConfig(level=logging.INFO)

from communications.pc import PC

if __name__ == "__main__":
    server = PC()
    server.connect()
    try:
        while True:
            message = server.receive()
            if message is None:
                break
            logging.info(f"RPi Received!:: {message}")
            # optional echo/ack:
            server.send(f"ack: {message}")
    finally:
        server.disconnect()
