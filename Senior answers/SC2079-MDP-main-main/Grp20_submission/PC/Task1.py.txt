    def pruneStaleDetections(self, now_ns: int) -> None:
        """Drop detections that are too old to be matched safely."""
        keep_for_ns = int(max(self.lead, self.max_match_age_ns))
        cutoff = now_ns - keep_for_ns
        stale_ids = [
            img_id for img_id, (_, last_seen) in self.img_time_dict.items()
            if last_seen < cutoff
        ]
        for img_id in stale_ids:
            del self.img_time_dict[img_id]

    def getOverlapInterval(self, img_id, timestamp, first_seen, last_seen):
        """
        Calculate the overlap interval between the search interval and the image's seen interval.
        Returns the overlap duration in nanoseconds.
        """
        if img_id not in self.img_time_dict:
            return 0
        
        search_interval = (timestamp - self.lead, timestamp + self.lag)
        img_interval = (first_seen, last_seen)
        overlap_start = max(search_interval[0], img_interval[0])
        overlap_end = min(search_interval[1], img_interval[1])
        overlap = max(0, overlap_end - overlap_start)
        return overlap

    def sendMatchedPair(self, obstacle_id, img_id):
        """
        Match the detected image with the given obstacle ID and send the information to the PC.
        """
        logging.info(f"Mapped image {img_id} to obstacle {obstacle_id}.")
        self.ids_to_stitch.append(img_id)
        logging.info(f"{len(self.ids_to_stitch)} images queued for stitching.")
        
        # Format: OBJECT,<obstacle_id>,<confidence_level>,<image_id>
        message_content = f"OBJECT,{obstacle_id},{self.stitching_img_dict[img_id][0]},{img_id}\n"
        self.IMG_BLACKLIST.append(str(img_id))
        logging.info(f"Dispatching message: {message_content}")
        self.sock.send(message_content.encode("utf-8"))
        
        # Clear stale images from stitching dict
        self.stitching_img_dict = {
            k: v for k, v in self.stitching_img_dict.items()
            if str(k) in self.IMG_BLACKLIST  # keep only those that are matched
        }
        
    def handleMessage(self, data_str: str) -> None:
        """Process a single complete message from RPi."""
        logging.info(f"Incoming RPi message: {data_str}")

        if data_str.startswith("OBSTACLES"):
            obstacles = self.parseObstacles(data_str)
            logging.info(f"Obstacle payload parsed: {obstacles}")
            logging.info(f"Starting route computation for {len(obstacles)} obstacle(s)...")
            _t0 = time_ns()

            path = pathfinding(obstacles)

            _elapsed = (time_ns() - _t0) / 1e9
            logging.info(f"Route computation completed in {_elapsed:.3f}s.")
            commands = path['commands']
            segments = self.segmentCommands(commands)
            segments['dirs'] = self.getDirections(path)
            logging.info(f"Command segments prepared: {segments}")
            if obstacles and not segments.get('obstacle_ids'):
                logging.warning(
                    "No reachable viewpoints were found for the current map; "
                    "returning a stop-only path."
                )
            self.sock.send(f"PATH,{json.dumps(segments)}\n".encode("utf-8"))
            logging.info("Path payload sent to RPi.")

        elif data_str.startswith("DETECT"):
            obstacle_id = data_str.split(",")[1].strip()
            timestamp = time_ns()
            self.pruneStaleDetections(timestamp)

            # Primary: overlap-based match
            max_overlap = 0
            max_img_id = None
            recent_candidates = []
            for img_id, (first_seen, last_seen) in self.img_time_dict.items():
                age_ns = timestamp - last_seen
                if age_ns > self.max_match_age_ns:
                    continue
                recent_candidates.append((age_ns, img_id))
                overlap = self.getOverlapInterval(img_id, timestamp, first_seen, last_seen)
                logging.info(f"Overlap check: {overlap}, current best: {max_overlap}")
                if overlap > 0 and overlap >= max_overlap:
                    logging.info(f"Updating best overlap to {overlap}")
                    max_overlap = overlap
                    max_img_id = img_id

            # Optional fallback: pick the most recently seen image that is still fresh.
            if max_img_id is None and self.allow_recent_fallback and recent_candidates:
                age_ns, max_img_id = min(recent_candidates, key=lambda x: x[0])
                logging.info(
                    "No overlap match; fallback selected recent image %s, seen %.2fs before DETECT",
                    max_img_id,
                    age_ns / 1e9,
                )

            if max_img_id is not None:
                self.sendMatchedPair(obstacle_id, max_img_id)
                del self.img_time_dict[max_img_id]
            else:
                logging.info(
                    "No valid image match was found for obstacle %s (fresh detections=%d).",
                    obstacle_id,
                    len(recent_candidates),
                )

        elif data_str.startswith("STITCH"):
            self.stitch_len = int(data_str.split(",")[1].strip())

            if len(self.ids_to_stitch) < self.stitch_len:
                logging.info("Stitch request received; waiting for completion...")
                self.should_stitch = True
                sleep(self.lag * 2e-9)
                stitchImages(self.ids_to_stitch, self.stitching_img_dict, filename=self.filename)
            else:
                logging.info("All required images are present; stitching now...")
                self.stream_listener.close()
                stitchImages(self.ids_to_stitch, self.stitching_img_dict, filename=self.filename)

    def pcReceive(self) -> None:
        self.connect()
        logging.info("PC socket connection started successfully.")
        _buf = ""

        while not self.exit:
            try:
                chunk = self.sock.recv(4096).decode("utf-8")
                if not chunk:
                    logging.info("Peer closed the connection.")
                    if self.ids_to_stitch:
                        stitchImages(self.ids_to_stitch, self.stitching_img_dict, filename=self.filename)
                    break
                _buf += chunk
                # Process every complete newline-terminated message
                while "\n" in _buf:
                    msg, _buf = _buf.split("\n", 1)
                    msg = msg.strip()
                    if msg:
                        self.handleMessage(msg)
            except OSError as e:
                logging.error(f"Receive loop error: {e}")
                break
            
    def startStream(self):
        self.stream_listener = StreamListener(weights=self.model)
        self.stream_listener.startStreamRead(
            onResult=self.onResult,
            on_disconnect=self.disconnect,
            conf_threshold=self.conf_threshold,
            show_video=False
        )
        logging.info("Stream started.")

    def connect(self, retries: int = 10, delay: float = 1.0):
        last_err = None
        for attempt in range(1, retries + 1):
            try:
                logging.info(f"Connecting to {self.host}:{self.port} (attempt {attempt}/{retries})")
                self.sock = socket.create_connection((self.host, self.port), timeout=self.timeout)
                self.sock.setsockopt(socket.IPPROTO_TCP, socket.TCP_NODELAY, 1)
                self.sock.settimeout(None)
                logging.info("Connection established.")
                return
            except OSError as e:
                last_err = e
                logging.warning(f"Connection attempt failed: {e}; retrying in {delay}s")
                sleep(delay)
        raise ConnectionError(f"Unable to connect to {self.host}:{self.port}") from last_err
        
    def disconnect(self):
        if self.sock:
            self.sock.close()
            self.sock = None
            self.exit = True
        
    def startTask1(self):        
        self.pc_receive_thread = threading.Thread(target=self.pcReceive, daemon=True)
        print("Starting Task 1 service...")
        self.stream_listener = threading.Thread(target=self.startStream, daemon=True)
        self.stream_listener.start()
        self.pc_receive_thread.start()
        self.pc_receive_thread.join()
                
    def parseObstacles(self, data_str):
        parts = data_str.split("OBSTACLES,")
        obstacle = json.loads(parts[1])
        return obstacle
    
    def segmentCommands(self, commands):
        # segment based on 'SNAP' commands into a list of lists
        segments = []
        current_segment = []
        obstacle_ids = []
        for cmd in commands:
            dist = cmd[2:]
            if cmd.startswith("BW"):
                cmd = "R"
                cmd += dist
            elif cmd.startswith("BL"):
                cmd = "RL"
                cmd += dist
            elif cmd.startswith("BR"):
                cmd = "RR"
                cmd += dist
            elif cmd.startswith("FW"):
                cmd = "F"
                cmd += dist
            elif cmd.startswith("FIN"):
                cmd = "S"
            if cmd.startswith("SNAP"):
                # SNAP<obstacle_id>[_L|_C|_R]
                snap_payload = cmd[4:]
                snap_id = snap_payload.split("_", 1)[0]
                obstacle_ids.append(snap_id)
                segments.append(current_segment)
                current_segment = []
                continue
            current_segment.append(cmd)
        if current_segment:
            segments.append(current_segment)
        return {"obstacle_ids": obstacle_ids, "segments": segments}
    
    def directionToName(self, d):
        """Return 'NORTH', 'SOUTH', etc. Works if d is an enum, int, or string."""
        # Already an enum with .name
        if hasattr(d, "name"):
            return d.name
        # Int -> enum name if possible
        try:
            return Direction(int(d)).name
        except Exception:
            pass
        # Fallback: if it's already a string
        if isinstance(d, str):
            return d.upper()
        # Last resort
        return str(d)
